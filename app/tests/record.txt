# SING mode: every character sings one note at a time, and the keys held are
# shared out among the choir. REC records what is sung into the loop (2 bars at
# 120 BPM: 4 s), played back pass after pass; the metronome counts the beats
# while recording; EDIT undoes a pass at a time, across the characters.
wait 600
tap GLO
expect mode 1
# Alone, a character is a mono synth: the last key wins, and when it goes the
# one still held sounds again.
key 0 down
wait 100
key 2 down
wait 100
expect singing 1
expect sung 2
key 2 up
wait 100
expect singing 1
expect sung 0
key 0 up
wait 600
expect singing 0
# The choir: HERO (selected), GUARD and SISTER.
turn SELECT 1
tap SAVE
turn SELECT 1
tap SAVE
turn SELECT -2
expect cast 0
wait 1500
# A chord: one letter for each of the three; a fourth key moves the singer of
# the oldest; let it go and that singer takes its key back.
key 4 down
key 7 down
key 11 down
wait 100
expect singing 3
key 14 down
wait 100
expect singing 3
expect sung 14
key 14 up
wait 100
expect singing 3
key 4 up
key 7 up
key 11 up
wait 300
expect singing 0
# REC starts the loop and records; the loop holds nothing yet.
tap REC
expect recording 1
expect loop 1
expect metronome 8
wait 500
key 0 down
wait 400
key 0 up
wait 50
expect take 1
expect passes 1
wait 600
expect sounding 0
shot record-play.png
# The next pass: HERO's letter sounds again with no key held, then lets go.
wait 3100
expect sounding 1
wait 900
expect sounding 0
# Overdub a chord: E, H and L go to HERO, GUARD and SISTER.
key 4 down
key 7 down
key 11 down
wait 300
expect singing 3
key 4 up
key 7 up
key 11 up
wait 50
expect take 2
expect passes 2
expect takes 4
# REC again stops recording (and the metronome); the loop keeps playing.
tap REC
expect recording 0
expect metronome 0
# LOOP mode: a lane for each singer.
tap GLO
expect mode 2
expect recording 0
shot record-lanes.png
wait 3000
expect takes 4
# EDIT undoes the last pass (the chord, all three singers), then the first,
# in LOOP mode as in SING.
tap EDIT
expect take 1
expect takes 1
expect passes 1
tap EDIT
expect take 0
expect takes 0
expect passes 0
