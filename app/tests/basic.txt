wait 600
shot boot.png

expect cast 0
# (The FM-1 boots in SING; this script starts in TYPE: SELECT back one.)
turn SELECT -1
wait 50
# The first line ("Our princess is in another castle!") cleared: EDIT held.
expect length 34
press EDIT
wait 1100
release EDIT
wait 50
expect length 0
# TYPE: keys type the line; a vowel speaks as it is typed.
tapkey 7
tapkey 4
expect length 2
wait 60
expect sounding 1
wait 400
shot typed.png
# PRESET: the next character greets; the line is everyone's and stays.
turn PRESET 1
wait 300

expect length 2
# SELECT: SING mode; a held key sings the line's first syllable.
turn SELECT 1
expect mode 0
key 0 down
wait 300
expect sounding 1
expect lyric 0
shot singing.png
key 0 up
wait 600
# PLAY: the loop runs (always looping); SELECT on to LOOP mode shows its lanes.
tap PLAY
wait 100
expect song 2

turn SELECT 1
wait 30
